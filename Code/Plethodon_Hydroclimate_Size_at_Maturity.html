Minimum SVL per site and sex, combined with site environmental data

# Smallest mature SVL per site and sex
Plethodon_sites_simplified <- all_mature_plethodon %>%
  group_by(site, maturity_status) %>%
  summarise(svl_mm = min(svl_mm), .groups = "drop") %>%
  mutate(sex = if_else(maturity_status == "mature_male", "male", "female"))

# Keeps only sites with measured mature individuals
Modeling_Data <- Plethodon_Study_Sites %>%
  inner_join(Plethodon_sites_simplified, by = "site", relationship = "one-to-many")

Prepping the Plethodon data for modeling

# Standardizes each site once
scale_by_site <- function(x, site) {
  once <- !duplicated(site)
  (x - mean(x[once])) / sd(x[once])
}

Modeling_Data <- Modeling_Data %>%
  mutate(
    # Folds aspect so 0 = north-facing and 180 = south-facing
    Aspect_folded = 180 - abs(Aspect - 180),

    # Standardizes the covariates
    MDT_z    = scale_by_site(DaymetMDTC.Apr.Oct.2011.2020, site),
    MDP_z    = scale_by_site(DaymetMDPmm.Apr.Oct.2011.2020, site),
    MDVPD_z  = scale_by_site(DaymetMDVPD.Apr.Oct.2011.2020, site),
    Aspect_z = scale_by_site(Aspect_folded, site),
    Slope_z  = scale_by_site(Slope, site),

    # Sets male as the reference level
    sex = relevel(factor(sex), ref = "male"))

Site_Data <- Modeling_Data %>%
  distinct(site, .keep_all = TRUE)

# Back-transformation values

MDVPD_sd   <- sd(Site_Data$DaymetMDVPD.Apr.Oct.2011.2020)

Slope_mean <- mean(Site_Data$Slope)

Calculating shared limits

# Both MDVPD plots begin and end with the Coweeta prediction range for comparison of the Coweeta plot to the broader regional plot.
MDVPD_limits <- range(MDVPD_pred_coweeta$MDVPD)

# Both slope plots cover the combined Coweeta and all-sites range
Slope_limits <- range(Slope_pred_coweeta$Slope, Slope_pred_all_sites$Slope)

# All four plots share one SVL axis that fits every confidence interval
SVL_limits <- range(
  MDVPD_pred_coweeta$conf.low,   MDVPD_pred_coweeta$conf.high,
  Slope_pred_coweeta$conf.low,   Slope_pred_coweeta$conf.high,
  MDVPD_pred_all_sites$conf.low, MDVPD_pred_all_sites$conf.high,
  Slope_pred_all_sites$conf.low, Slope_pred_all_sites$conf.high)

# Adding some space above and below the confidence intervals
SVL_padding <- diff(SVL_limits) * 0.05
SVL_limits <- c(SVL_limits[1] - SVL_padding, SVL_limits[2] + SVL_padding)
